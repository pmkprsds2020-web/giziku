   124→
   125→export function MealPlanView() {
   126→  const { data: patients } = usePatients();
   127→  const { activePatientId } = useCareLiviaStore();
   128→  const [selectedPatientId, setSelectedPatientId] = React.useState<string>(
   129→    activePatientId || "",
   130→  );
   131→  const [activePresetId, setActivePresetId] = React.useState<string | null>(null);
   132→  const [activePresetName, setActivePresetName] = React.useState<string | null>(null);
   133→  const [previewData, setPreviewData] = React.useState<any>(null);
   134→  const [showSaveDialog, setShowSaveDialog] = React.useState(false);
   135→  const [saveName, setSaveName] = React.useState("");
   136→
   137→  React.useEffect(() => {
   138→    if (activePatientId && !selectedPatientId) {
   139→      setSelectedPatientId(activePatientId);
   140→    }
   141→  }, [activePatientId, selectedPatientId]);
   142→
   143→  const { data: plans, isLoading } = useMealPlans(selectedPatientId || undefined);
   144→  const { data: presets } = usePresets(selectedPatientId || undefined);
   145→  const generateMut = useGenerateMealPlan();
   146→  const previewMut = usePreviewIsiPiringku();
   147→  const saveMut = useCreateSavedMealPlan();
   148→
   149→  const latestPlan = plans?.[0];
   150→  const selectedPatient = patients?.find((p) => p.id === selectedPatientId);
   151→  const patientPresets = (presets || []).filter((p: any) => !p.isTemplate);
   152→  const templatePresets = (presets || []).filter((p: any) => p.isTemplate);
   153→
   154→  // Sync active preset from latest meal plan
   155→  React.useEffect(() => {
   156→    if (latestPlan?.presetId && !activePresetId) {
   157→      setActivePresetId(latestPlan.presetId);
   158→      setActivePresetName(latestPlan.preset?.name || null);
   159→    }
   160→  }, [latestPlan, activePresetId]);
   161→
   162→  const handleGenerate = async () => {
   163→    if (!selectedPatientId) {
   164→      toast.error("Pilih pasien terlebih dahulu");
   165→      return;
   166→    }
   167→    try {
   168→      // Use preview endpoint (no persistence) for live view
   169→      const result = await previewMut.mutateAsync({
   170→        patientId: selectedPatientId,
   171→        presetId: activePresetId || undefined,
   172→      });
   173→      setPreviewData(result);
   174→      toast.success(
   175→        activePresetName
   176→          ? `Meal plan Isi Piringku dibuat dengan preset "${activePresetName}"`
   177→          : "Meal plan Isi Piringku berhasil dibuat",
   178→      );
   179→    } catch (e: any) {
   180→      toast.error(e.message || "Gagal membuat meal plan");
   181→    }
   182→  };
   183→
   184→  const handlePersist = async () => {
   185→    if (!selectedPatientId) return;
   186→    try {
   187→      await generateMut.mutateAsync({
   188→        patientId: selectedPatientId,
   189→        presetId: activePresetId || undefined,
   190→      });
   191→      toast.success("Meal plan disimpan ke database");
   192→    } catch (e: any) {
   193→      toast.error(e.message || "Gagal menyimpan meal plan");
   194→    }
   195→  };
   196→
   197→  const handleSaveAsTemplate = async () => {
   198→    if (!selectedPatientId || !previewData) return;
   199→    try {
   200→      await saveMut.mutateAsync({
   201→        patientId: selectedPatientId,
   202→        name: saveName || `Meal Plan ${new Date().toLocaleDateString("id-ID")}`,
   203→        date: new Date().toISOString(),
   204→        totalCal: previewData.plan.totals.cal,
   205→        totalProtein: previewData.plan.totals.protein,
   206→        totalFat: previewData.plan.totals.fat,
   207→        totalCarb: previewData.plan.totals.carb,
   208→        totalFiber: previewData.plan.totals.fiber,
   209→        totalSodium: previewData.plan.totals.sodium,
   210→        compliance: previewData.plan.overallCompliance,
   211→        notes: `Compliance: ${previewData.plan.overallTierLabel}`,
   212→        items: previewData.plan.items.map((i: any) => ({
   213→          slot: i.slot,
   214→          foodId: i.foodId,
   215→          amount: i.amount,
   216→          cal: i.cal,
   217→          protein: i.protein,
   218→          fat: i.fat,
   219→          carb: i.carb,
   220→          fiber: i.fiber,
   221→          sodium: i.sodium,
   222→        })),
   223→      });
   224→      toast.success(`Meal plan "${saveName}" disimpan ke library`);
   225→      setShowSaveDialog(false);
   226→      setSaveName("");
   227→    } catch (e: any) {
   228→      toast.error(e.message || "Gagal menyimpan meal plan");
   229→    }
   230→  };
   231→
   232→  const handleSelectPreset = (preset: any) => {
   233→    setActivePresetId(preset.id);
   234→    setActivePresetName(preset.name);
   235→    toast.success(`Preset "${preset.name}" dipilih — klik Generate untuk membuat Meal Plan baru`);
   236→  };
   237→
   238→  const plan = previewData?.plan;
   239→  const calorieResult = previewData?.calorieResult;
   240→  const aiReasoning = previewData?.aiReasoning;
   241→  const patientInfo = previewData?.patient;
   242→  const targets = previewData?.targets;
   243→
   244→  return (
   245→    <div className="space-y-6">
   246→      <PageHeader
   247→        title="AI Meal Plan — Isi Piringku"
   248→        subtitle="Pedoman Gizi Seimbang Kemenkes RI: 4 kelompok makanan di setiap makan utama"
   249→        icon={Utensils}
   250→        actions={
   251→          <>
   252→            <Button
   253→              variant="outline"
   254→              size="sm"
   255→              onClick={() => {
   256→                setPreviewData(null);
   257→                setActivePresetId(null);
   258→                setActivePresetName(null);
   259→              }}
   260→            >
   261→              <X className="mr-1.5 h-4 w-4" /> Reset
   262→            </Button>
   263→            <Button
   264→              onClick={handleGenerate}
   265→              disabled={previewMut.isPending || !selectedPatientId}
   266→              size="sm"
   267→            >
   268→              <Sparkles className="mr-1.5 h-4 w-4" />
   269→              {previewMut.isPending ? "Menyusun…" : "Generate Isi Piringku"}
   270→            </Button>
   271→          </>
   272→        }
   273→      />
   274→
   275→      {/* Patient + Preset selector */}
   276→      <Card className="border-border/60">
   277→        <CardContent className="p-5">
   278→          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   279→            <div>
   280→              <Label className="text-xs font-medium text-muted-foreground">
   281→                <User className="mr-1 inline h-3.5 w-3.5" /> Pasien
   282→              </Label>
   283→              <Select
   284→                value={selectedPatientId}
   285→                onValueChange={setSelectedPatientId}
   286→              >
   287→                <SelectTrigger className="mt-1.5">
   288→                  <SelectValue placeholder="Pilih pasien" />
   289→                </SelectTrigger>
   290→                <SelectContent>
   291→                  {(patients || []).map((p: any) => (
   292→                    <SelectItem key={p.id} value={p.id}>
   293→                      {p.name} — {p.mrn}
   294→                    </SelectItem>
   295→                  ))}
   296→                </SelectContent>
   297→              </Select>
   298→            </div>
   299→
   300→            <div>
   301→              <Label className="text-xs font-medium text-muted-foreground">
   302→                <Target className="mr-1 inline h-3.5 w-3.5" /> Preset Aktif (opsional)
   303→              </Label>
   304→              <Select
   305→                value={activePresetId || "__none__"}
   306→                onValueChange={(v) => {
   307→                  if (v === "__none__") {
   308→                    setActivePresetId(null);
   309→                    setActivePresetName(null);
   310→                  } else {
   311→                    const p = (presets || []).find((x: any) => x.id === v);
   312→                    if (p) handleSelectPreset(p);
   313→                  }
   314→                }}
   315→              >
   316→                <SelectTrigger className="mt-1.5">
   317→                  <SelectValue placeholder="Tanpa preset (pakai target CareLivia)" />
   318→                </SelectTrigger>
   319→                <SelectContent>
   320→                  <SelectItem value="__none__">— Tanpa preset —</SelectItem>
   321→                  {patientPresets.length > 0 && (
   322→                    <>
   323→                      <div className="px-2 py-1 text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   324→                        Preset Pasien
   325→                      </div>
   326→                      {patientPresets.map((p: any) => (
   327→                        <SelectItem key={p.id} value={p.id}>
   328→                          {p.name} ({p.totalCal} kcal)
   329→                        </SelectItem>
   330→                      ))}
   331→                    </>
   332→                  )}
   333→                  {templatePresets.length > 0 && (
   334→                    <>
   335→                      <div className="px-2 py-1 text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   336→                        Template
   337→                      </div>
   338→                      {templatePresets.map((p: any) => (
   339→                        <SelectItem key={p.id} value={p.id}>
   340→                          {p.name} ({p.totalCal} kcal)
   341→                        </SelectItem>
   342→                      ))}
   343→                    </>
   344→                  )}
   345→                </SelectContent>
   346→              </Select>
   347→            </div>
   348→
   349→            <div>
   350→              <Label className="text-xs font-medium text-muted-foreground">
   351→                <Activity className="mr-1 inline h-3.5 w-3.5" /> Diagnosis Utama
   352→              </Label>
   353→              <div className="mt-1.5 flex flex-wrap gap-1">
   354→                {selectedPatient?.diagnoses?.length ? (
   355→                  selectedPatient.diagnoses.map((raw: any, idx: number) => {
   356→                    const d: string = typeof raw === "string" ? raw : (raw?.type ?? "");
   357→                    if (!d) return null;
   358→                    return (
   359→                      <Badge key={`${d}-${idx}`} variant="secondary" className="text-xs">
   360→                        {DIAGNOSIS_ADJUSTMENTS[d as DiagnosisType]?.label || d}
   361→                      </Badge>
   362→                    );
   363→                  })
   364→                ) : (
   365→                  <Badge variant="outline" className="text-xs">Umum</Badge>
   366→                )}
   367→              </div>
   368→            </div>
   369→          </div>
   370→
   371→          {activePresetName && (
   372→            <Alert className="mt-4 border-violet-500/30 bg-violet-500/5">
   373→              <Target className="h-4 w-4 text-violet-600" />
   374→              <AlertTitle className="text-sm">Preset aktif: {activePresetName}</AlertTitle>
   375→              <AlertDescription className="text-xs">
   376→                Target akan di-override dengan nilai preset. Klik Generate untuk menyusun ulang.
   377→              </AlertDescription>
   378→            </Alert>
   379→          )}
   380→        </CardContent>
   381→      </Card>
   382→
   383→      {/* Loading state */}
   384→      {previewMut.isPending && !plan && (
   385→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   386→          {Array.from({ length: 3 }).map((_, i) => (
   387→            <Skeleton key={i} className="h-72 w-full rounded-xl" />
   388→          ))}
   389→        </div>
   390→      )}
   391→
   392→      {/* Empty state */}
   393→      {!plan && !previewMut.isPending && (
   394→        <EmptyState
   395→          icon={Utensils}
   396→          title="Belum ada Meal Plan"
   397→          description="Pilih pasien lalu klik Generate untuk menyusun rencana makan sesuai Pedoman Isi Piringku Kemenkes RI. Setiap makan utama akan berisi 4 kelompok: makanan pokok, lauk pauk, sayuran, dan buah."
   398→          action={
   399→            <Button onClick={handleGenerate} disabled={!selectedPatientId}>
   400→              <Sparkles className="mr-1.5 h-4 w-4" /> Generate Sekarang
   401→            </Button>
   402→          }
   403→        />
   404→      )}
   405→
   406→      {/* MAIN CONTENT — Isi Piringku plan */}
   407→      {plan && calorieResult && (
   408→        <>
   409→          {/* Top summary stats */}
   410→          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   411→            <StatCard
   412→              label="Target Energi"
   413→              value={calorieResult.targetCalorie.toLocaleString("id-ID")}
   414→              unit="kcal"
   415→              icon={Target}
   416→              color="primary"
   417→              sublabel={`BMI ${calorieResult.bmi} (${calorieResult.bmiLabel})`}
   418→            />
   419→            <StatCard
   420→              label="Energi Aktual"
   421→              value={plan.totals.cal.toLocaleString("id-ID")}
   422→              unit="kcal"
   423→              icon={TrendingUp}
   424→              color="emerald"
   425→              sublabel={`${Math.round((plan.totals.cal / calorieResult.targetCalorie) * 100)}% dari target`}
   426→            />
   427→            <StatCard
   428→              label="Compliance Isi Piringku"
   429→              value={plan.overallCompliance}
   430→              unit="%"
   431→              icon={CheckCircle2}
   432→              color={plan.overallCompliance >= 90 ? "emerald" : plan.overallCompliance >= 70 ? "amber" : "rose"}
   433→              sublabel={`${COMPLIANCE_TIER_ICON[plan.overallTier]} ${plan.overallTierLabel}`}
   434→            />
   435→            <StatCard
   436→              label="Distribusi Makro"
   437→              value={`P${Math.round((plan.totals.protein * 4 / plan.totals.cal) * 100)}/L${Math.round((plan.totals.fat * 9 / plan.totals.cal) * 100)}/K${Math.round((plan.totals.carb * 4 / plan.totals.cal) * 100)}`}
   438→              icon={Activity}
   439→              color="violet"
   440→              sublabel={`Protein ${plan.totals.protein}g · Lemak ${plan.totals.fat}g · Karbo ${plan.totals.carb}g`}
   441→            />
   442→          </div>
   443→
   444→          {/* Action bar */}
   445→          <div className="flex flex-wrap items-center gap-2">
   446→            <Button onClick={handleGenerate} variant="outline" size="sm" disabled={previewMut.isPending}>
   447→              <RefreshCw className="mr-1.5 h-4 w-4" />
   448→              {previewMut.isPending ? "Menyusun ulang…" : "Generate Ulang"}
   449→            </Button>
   450→            <Button onClick={handlePersist} variant="default" size="sm" disabled={generateMut.isPending}>
   451→              <Save className="mr-1.5 h-4 w-4" />
   452→              {generateMut.isPending ? "Menyimpan…" : "Simpan ke Database"}
   453→            </Button>
   454→            <Button onClick={() => setShowSaveDialog(true)} variant="outline" size="sm">
   455→              <Bookmark className="mr-1.5 h-4 w-4" /> Simpan ke Library
   456→            </Button>
   457→            {patientInfo && (
   458→              <Badge variant="outline" className="ml-auto text-xs">
   459→                <User className="mr-1 h-3 w-3" />
   460→                {patientInfo.name} · {patientInfo.mrn}
   461→              </Badge>
   462→            )}
   463→          </div>
   464→
   465→          {/* ROTATION WARNINGS */}
   466→          {plan.rotationWarnings.length > 0 && (
   467→            <Alert className="border-amber-500/30 bg-amber-500/5">
   468→              <RotateCw className="h-4 w-4 text-amber-600" />
   469→              <AlertTitle className="text-sm">Peringatan Variasi Menu (Rotasi 14 Hari)</AlertTitle>
   470→              <AlertDescription className="text-xs">
   471→                <ul className="list-disc pl-4 space-y-0.5">
   472→                  {plan.rotationWarnings.map((w: string, i: number) => (
   473→                    <li key={i}>{w}</li>
   474→                  ))}
   475→                </ul>
   476→              </AlertDescription>
   477→            </Alert>
   478→          )}
   479→
   480→          {/* PLATE VISUALIZATION — 3 main meals */}
   481→          <SectionCard
   482→            title="Visualisasi Isi Piringku"
   483→            description="Setiap makan utama mengikuti proporsi: 2/3 makanan pokok + 1/3 lauk pauk (setengah piring pertama); 2/3 sayuran + 1/3 buah (setengah piring kedua)"
   484→          >
   485→            <div className="grid gap-6 sm:grid-cols-2 lg:grid-cols-3">
   486→              {plan.slotSummaries
   487→                .filter((s: any) => MAIN_SLOTS.includes(s.slot))
   488→                .map((s: any) => (
   489→                  <PlateMealCard key={s.slot} summary={s} />
   490→                ))}
   491→            </div>
   492→          </SectionCard>
   493→
   494→          {/* DETAILED MEAL ITEMS BY SLOT */}
   495→          <SectionCard
   496→            title="Daftar Menu Lengkap"
   497→            description="Klik item untuk melihat alternatif pengganti setara nutrisi"
   498→          >
   499→            <div className="space-y-3">
   500→              {plan.slotSummaries.map((s: any) => (
   501→                <SlotItemsCard key={s.slot} summary={s} />
   502→              ))}
   503→            </div>
   504→          </SectionCard>
   505→
   506→          {/* NUTRITION VALIDATION TABLE */}
   507→          <SectionCard
   508→            title="Validasi Kecukupan Gizi"
   509→            description="Target terpenuhi jika 95-105% dari kebutuhan harian (Kemenkes RI)"
   510→          >
   511→            <NutritionValidationTable validation={plan.validation} />
   512→          </SectionCard>
   513→
   514→          {/* AI REASONING */}
   515→          {aiReasoning && (
   516→            <SectionCard
   517→              title="Evaluasi AI CareLivia"
   518→              description="Analisis klinis berbasis Pedoman Isi Piringku & diagnosis pasien"
   519→            >
   520→              <div className="rounded-lg border border-violet-500/20 bg-gradient-to-br from-violet-500/5 to-primary/5 p-4">
   521→                <div className="mb-2 flex items-center gap-2">
   522→                  <div className="flex h-7 w-7 items-center justify-center rounded-full bg-violet-500/10">
   523→                    <Brain className="h-3.5 w-3.5 text-violet-600" />
   524→                  </div>
   525→                  <span className="text-sm font-semibold text-violet-700 dark:text-violet-300">
   526→                    CareLivia AI Nutritionist
   527→                  </span>
   528→                </div>
   529→                <p className="whitespace-pre-line text-sm leading-relaxed text-foreground/90">
   530→                  {aiReasoning}
   531→                </p>
   532→              </div>
   533→            </SectionCard>
   534→          )}
   535→
   536→          {/* GROUP COVERAGE SUMMARY */}

   886→  return (
   887→    <div className="overflow-x-auto">
   888→      <table className="w-full text-sm">
   889→        <thead>
   890→          <tr className="border-b border-border/60 text-left text-xs text-muted-foreground">
