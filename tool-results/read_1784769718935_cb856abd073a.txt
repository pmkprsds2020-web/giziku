   398→   398→export async function getServerClient(): Promise<{
   399→   399→  client: SupabaseClient;
   400→   400→  session: { user: any } | null;
   401→   401→}> {

   411→   411→// FOODS — Public reads (anon OK), writes require auth
   412→   412→// ---------------------------------------------------------------------
   413→   413→export async function supabaseListFoods(params?: {
   414→   414→  search?: string;
   415→   415→  categorySlug?: string;
   416→   416→  limit?: number;
   417→   417→  offset?: number;
   418→   418→}): Promise<{ data: any[]; total: number }> {
   419→   419→  const { client } = await getServerClient();
   420→   420→  const limit = params?.limit ?? 100;
   421→   421→  const offset = params?.offset ?? 0;
   422→   422→
   423→   423→  let query = client
   424→   424→    .from("foods")
   425→   425→    .select("*, food_categories!foods_category_id_fkey(*)", { count: "exact" })
   426→   426→    .is("deleted_at", null)
   427→   427→    .eq("approved", true)
   428→   428→    .range(offset, offset + limit - 1)
   429→   429→    .order("name");
   430→   430→
   431→   431→  if (params?.search) {
   432→   432→    query = query.or(`name.ilike.%${params.search}%,english_name.ilike.%${params.search}%,tags.ilike.%${params.search}%`);
   433→   433→  }
   434→   434→
   435→   435→  // Category filter requires subquery — skip for now, filter in memory
   436→   436→  const { data, error, count } = await query;
   437→   437→
   438→   438→  if (error) {
   439→   439→    console.error("[Supabase] listFoods error:", error);
   440→   440→    return { data: [], total: 0 };
   441→   441→  }
   442→   442→
   443→   443→  let rows = (data || []).map((row: any) =>
   444→   444→    foodFromSupabase(row, row.food_categories),
   445→   445→  );
   446→   446→
   447→   447→  // In-memory category filter
   448→   448→  if (params?.categorySlug && params.categorySlug !== "all") {
   449→   449→    rows = rows.filter((f: any) => f.category?.slug === params.categorySlug);
   450→   450→  }
   451→   451→
   452→   452→  return { data: rows, total: count ?? rows.length };
   453→   453→}

   455→   455→export async function supabaseGetFood(id: string): Promise<any | null> {
   456→   456→  const { client } = await getServerClient();
   457→   457→  const { data, error } = await client
   458→   458→    .from("foods")
   459→   459→    .select("*, food_categories!foods_category_id_fkey(*)")
   460→   460→    .eq("id", id)
   461→   461→    .single();
   462→   462→
   463→   463→  if (error) {
   464→   464→    console.error("[Supabase] getFood error:", error);
   465→   465→    return null;
   466→   466→  }
   467→   467→

   678→   678→export async function supabaseCreateMealPlan(plan: any, items: any[]): Promise<{ data: any | null; error: string | null }> {
   679→   679→  const { client, session } = await getServerClient();

   681→   681→
   682→   682→  const planPayload = mealPlanToSupabase(plan);

   698→   698→    const itemPayloads = items.map((i) => {
   699→   699→      const payload = mealItemToSupabase(i);
   700→   700→      payload.meal_plan_id = planData.id;
   701→   701→      delete payload.id;
   702→   702→      return payload;

   710→   710→      console.error("[Supabase] createMealPlanItems error:", itemsError);

   713→   713→    }
   714→   714→  }

   716→   716→  // Fetch complete plan with items

   718→   718→    .from("meal_plans")
   719→   719→    .select("*, meal_plan_items(*, foods!meal_plan_items_food_id_fkey(*, food_categories!foods_category_id_fkey(*)))")

   725→   725→  );
   726→   726→
   727→   727→  return { data: mealPlanFromSupabase(fullPlan || planData, planItems), error: null };
