     5→// ---------------------------------------------------------------------
     6→// CareLivia — Frontend Supabase Data Layer (Browser Client)
     7→// ALL reads go directly to Supabase PostgreSQL via browser client.
     8→// No API routes for reads — eliminates 404s and ID mismatches.
     9→//
    10→// Writes also go directly to Supabase (RLS requires authenticated session).
    11→// Only AI/complex operations use API routes (meal-plan generate, exercise, etc.)
    12→// ---------------------------------------------------------------------
    13→
    14→let supabaseClient: ReturnType<typeof createClient> | null = null;
    15→
    16→function getSupabase() {
    17→  if (!supabaseClient) {
    18→    supabaseClient = createClient();
    19→  }
    20→  return supabaseClient;
    21→}

   134→export async function supabaseFetchFoods(params?: {
   135→  q?: string;
   136→  categoryId?: string;
   137→  highProtein?: boolean;
   138→  lowGi?: boolean;
   139→  lowSodium?: boolean;
   140→  highFiber?: boolean;
   141→}): Promise<{ foods: any[]; categories: any[] }> {
   142→  const supabase = getSupabase();
   143→
   144→  // Fetch foods with category
   145→  let query = supabase
   146→    .from("foods")
   147→    .select("*, food_categories(*)")
   148→    .is("deleted_at", null)
   149→    .eq("approved", true)
   150→    .order("name")
   151→    .limit(200);
   152→
   153→  if (params?.q) {
   154→    query = query.or(`name.ilike.%${params.q}%,english_name.ilike.%${params.q}%,tags.ilike.%${params.q}%`);
   155→  }
   156→
   157→  const { data: foods, error: foodsError } = await query;
   158→  if (foodsError) console.error("[frontend] fetchFoods error:", foodsError);
   159→
   160→  // Fetch categories
   161→  const { data: categories, error: catError } = await supabase
   162→    .from("food_categories")
   163→    .select("*")
   164→    .order("name");
   165→  if (catError) console.error("[frontend] fetchCategories error:", catError);
   166→
   167→  // Map foods to camelCase
   168→  let mapped = (foods || []).map((f: any) => ({
   169→    id: f.id,
   170→    name: f.name,
   171→    englishName: f.english_name,
   172→    categoryId: f.category_id,
   173→    category: f.food_categories ? {
   174→      id: f.food_categories.id,
   175→      name: f.food_categories.name,
   176→      slug: f.food_categories.slug,
   177→      icon: f.food_categories.icon,
   178→    } : null,
   179→    energy: f.energy,
   180→    protein: f.protein,
   181→    fat: f.fat,
   182→    carb: f.carb,
   183→    fiber: f.fiber,
   184→    sodium: f.sodium,
   185→    potassium: f.potassium,
   186→    calcium: f.calcium,
   187→    iron: f.iron,
   188→    gi: f.gi,
   189→    urt: f.urt,
   190→    urtGram: f.urt_gram,
   191→    price: f.price,
   192→    tags: f.tags,
   193→    source: f.source,
   194→  }));
   195→
   196→  // Apply filters
   197→  if (params?.categoryId) mapped = mapped.filter((f) => f.categoryId === params.categoryId);
   198→  if (params?.highProtein) mapped = mapped.filter((f) => f.protein >= 10);
   199→  if (params?.lowGi) mapped = mapped.filter((f) => f.gi < 55 && f.gi > 0);
   200→  if (params?.lowSodium) mapped = mapped.filter((f) => f.sodium <= 100);
   201→  if (params?.highFiber) mapped = mapped.filter((f) => f.fiber >= 3);
   202→
   203→  // Count foods per category
   204→  const countMap = new Map<string, number>();
   205→  for (const f of mapped) {
   206→    if (f.categoryId) countMap.set(f.categoryId, (countMap.get(f.categoryId) || 0) + 1);
   207→  }
   208→
   209→  const mappedCategories = (categories || []).map((c: any) => ({
   210→    id: c.id,
   211→    name: c.name,
   212→    slug: c.slug,
   213→    icon: c.icon,
   214→    _count: { foods: countMap.get(c.id) || 0 },
   215→  }));
   216→
   217→  return { foods: mapped, categories: mappedCategories };
   218→}

   223→export async function supabaseFetchDashboard(): Promise<any> {
   224→  const supabase = getSupabase();
   225→
   226→  // Count queries
   227→  const [patientsCount, foodsCount, mealPlansCount, foodRecordsData] = await Promise.all([
   228→    supabase.from("patients").select("id", { count: "exact", head: true }).is("deleted_at", null),
   229→    supabase.from("foods").select("id", { count: "exact", head: true }).is("deleted_at", null).eq("approved", true),
   230→    supabase.from("meal_plans").select("id", { count: "exact", head: true }).is("deleted_at", null).eq("status", "FINAL"),
   231→    supabase.from("food_records").select("id, cal, date", { count: "exact" }).limit(500),
   232→  ]);
   233→
   234→  // Fetch recent patients with diagnoses for summaries
   235→  const { data: patients } = await supabase
   236→    .from("patients")
   237→    .select("id, mrn, name, height, weight, diagnoses(*)")
   238→    .is("deleted_at", null)
   239→    .order("created_at", { ascending: false })
   240→    .limit(10);
   241→
   242→  // Fetch recent meal plans with patient info
   243→  const { data: recentPlansRaw } = await supabase
   244→    .from("meal_plans")
   245→    .select("id, date, total_cal, compliance, status, patients(name, mrn)")
   246→    .is("deleted_at", null)
   247→    .order("date", { ascending: false })
   248→    .limit(10);
   249→
   250→  const todayStart = new Date();
   251→  todayStart.setHours(0, 0, 0, 0);
   252→  const todayRecords = (foodRecordsData.data || []).filter((r: any) => new Date(r.date) >= todayStart);
   253→  const todayCalories = todayRecords.reduce((sum: number, r: any) => sum + (r.cal || 0), 0);
   254→
   255→  // Build patient summaries
   256→  const patientSummaries = (patients || []).map((p: any) => {
   257→    const bmi = p.height && p.weight && p.height > 0
   258→      ? p.weight / Math.pow(p.height / 100, 2)
   259→      : null;
   260→    const birthYear = parseInt(String(p.mrn).substring(0, 4)) || 1980;
   261→    return {
   262→      id: p.id,
   263→      mrn: p.mrn,
   264→      name: p.name,
   265→      bmi: bmi ? Math.round(bmi * 10) / 10 : null,
   266→      diagnoses: (p.diagnoses || []).filter((d: any) => d.active).map((d: any) => d.type),
   267→    };
   268→  });
   269→
   270→  // Build recent plans
   271→  const recentPlans = (recentPlansRaw || []).map((mp: any) => ({
   272→    id: mp.id,
   273→    date: mp.date,
   274→    totalCal: mp.total_cal,
   275→    compliance: mp.compliance,
   276→    status: mp.status,
   277→    patientName: mp.patients?.name || "Unknown",
   278→    patientMrn: mp.patients?.mrn || "",
   279→  }));
   280→
   281→  // Diagnosis distribution
   282→  const diagnosisDist: Record<string, number> = {};
   283→  for (const p of (patients || [])) {
   284→    for (const d of (p.diagnoses || [])) {
   285→      if (d.active) {
   286→        diagnosisDist[d.type] = (diagnosisDist[d.type] || 0) + 1;
   287→      }
   288→    }
   289→  }
   290→
   291→  return {
   292→    totalPatients: patientsCount.count ?? 0,
   293→    activeMealPlans: mealPlansCount.count ?? 0,
   294→    totalFoods: foodsCount.count ?? 0,
   295→    todayCalTotal: Math.round(todayCalories),
   296→    todayRecords: todayRecords.length,
   297→    patientSummaries,
   298→    recentPlans,
   299→    diagnosisDistribution: diagnosisDist,
   300→  };
   301→}

   306→export async function supabaseFetchWeightRecords(patientId: string): Promise<any> {
   307→  const supabase = getSupabase();
   308→  const { data, error } = await supabase
   309→    .from("weight_records")
   310→    .select("*")
   311→    .eq("patient_id", patientId)
   312→    .order("date", { ascending: true })
   313→    .limit(100);
   314→
   315→  if (error) {
   316→    console.error("[frontend] fetchWeightRecords error:", error);
   317→    return { records: [], summary: null, periodChanges: [], alerts: [] };
   318→  }
   319→
   320→  const records = (data || []).map((r: any) => ({
   321→    id: r.id,
   322→    patientId: r.patient_id,
   323→    date: r.date,
   324→    weight: r.weight,
   325→    height: r.height,
   326→    bmi: r.bmi,
   327→    bmiCategory: r.bmi_category,
   328→    weightChange: r.weight_change,
   329→    weightChangePct: r.weight_change_pct,
   330→    note: r.note,
   331→  }));
   332→
   333→  // Compute summary
   334→  const latest = records[records.length - 1] || null;
   335→  const first = records[0] || null;
   336→  const totalChange = latest && first ? Math.round((latest.weight - first.weight) * 10) / 10 : 0;
   337→  const totalPct = latest && first && first.weight > 0
   338→    ? Math.round(((latest.weight - first.weight) / first.weight) * 1000) / 10
   339→    : 0;
   340→
   341→  return {
   342→    records,
   343→    summary: latest ? {
   344→      latestWeight: latest.weight,
   345→      latestBMI: latest.bmi,
   346→      latestBMICategory: latest.bmiCategory,
   347→      latestDate: latest.date,
   348→      totalChange,
   349→      totalPct,
   350→      recordCount: records.length,
   351→    } : null,
   352→    periodChanges: [
   353→      { label: "7 hari", change: null, pct: null },
   354→      { label: "30 hari", change: null, pct: null },
   355→      { label: "3 bulan", change: null, pct: null },
   356→      { label: "6 bulan", change: null, pct: null },
   357→      { label: "1 tahun", change: null, pct: null },
   358→    ],
   359→    alerts: [],
   360→  };
   361→}

   363→export async function supabaseAddWeightRecord(data: {
   364→  patientId: string;
   365→  weight: number;
   366→  height?: number | null;
   367→  date?: string;
   368→  note?: string;
   369→}): Promise<any> {
   370→  const supabase = getSupabase();
   371→  const height = data.height ?? null;
   372→  const bmi = height ? Math.round((data.weight / Math.pow(height / 100, 2)) * 10) / 10 : null;
   373→  const bmiCategory = bmi
   374→    ? bmi < 18.5 ? "UNDERWEIGHT" : bmi < 23 ? "NORMAL" : bmi < 25 ? "OVERWEIGHT" : "OBESE"
   375→    : null;
   376→
   377→  const { data: record, error } = await supabase
   378→    .from("weight_records")
   379→    .insert({
   380→      patient_id: data.patientId,
   381→      date: data.date ?? new Date().toISOString(),
   382→      weight: data.weight,
   383→      height,
   384→      bmi,
   385→      bmi_category: bmiCategory,
   386→      note: data.note ?? "",
   387→    })
   388→    .select()
   389→    .single();
   390→
   391→  if (error) throw new Error(error.message);
   392→
   393→  // Update patient's current weight
   394→  await supabase
   395→    .from("patients")
   396→    .update({ weight: data.weight, height: height ?? undefined })
   397→    .eq("id", data.patientId);
   398→
   399→  return record;
   400→}

   546→export async function supabaseAddFoodRecord(data: {
   547→  patientId: string;
   548→  foodId: string;
   549→  slot: string;
   550→  amount: number;
   551→  consumed?: number;
   552→  date?: string;
   553→  notes?: string;
   554→}): Promise<any> {
   555→  const supabase = getSupabase();
   556→
   557→  // Fetch food to compute nutrition
   558→  const { data: food } = await supabase
   559→    .from("foods")
   560→    .select("*")
   561→    .eq("id", data.foodId)
   562→    .single();
   563→
   564→  if (!food) throw new Error("Makanan tidak ditemukan");
   565→
   566→  const ratio = data.amount / 100;
   567→  const consumedRatio = (data.consumed ?? 100) / 100;
   568→
   569→  const { data: record, error } = await supabase
   570→    .from("food_records")
   571→    .insert({
   572→      patient_id: data.patientId,
   573→      food_id: data.foodId,
   574→      slot: data.slot.toUpperCase(),
   575→      amount: data.amount,
   576→      consumed: data.consumed ?? 100,
   577→      date: data.date ?? new Date().toISOString(),
   578→      cal: food.energy * ratio * consumedRatio,
   579→      protein: food.protein * ratio * consumedRatio,
   580→      fat: food.fat * ratio * consumedRatio,
   581→      carb: food.carb * ratio * consumedRatio,
   582→      fiber: food.fiber * ratio * consumedRatio,
   583→      sodium: food.sodium * ratio * consumedRatio,
   584→      notes: data.notes ?? "",
   585→    })
   586→    .select()
   587→    .single();
   588→

   811→export async function supabaseCreateSavedMealPlan(data: any): Promise<any> {
   812→  const supabase = getSupabase();
   813→  const { data: plan, error } = await supabase
   814→    .from("saved_meal_plans")
   815→    .insert({
   816→      patient_id: data.patientId ?? null,
   817→      name: data.name,
   818→      total_cal: data.totalCal ?? 0,
   819→      total_protein: data.totalProtein ?? 0,
   820→      total_fat: data.totalFat ?? 0,
   821→      total_carb: data.totalCarb ?? 0,
   822→      total_fiber: data.totalFiber ?? 0,
   823→      total_sodium: data.totalSodium ?? 0,
   824→      notes: data.notes ?? "",
   825→    })
   826→    .select()
   827→    .single();
   828→
   829→  if (error) throw new Error(error.message);
   830→
   831→  if (data.items && data.items.length > 0) {
   832→    // Fetch food names
   833→    const foodIds = [...new Set(data.items.map((i: any) => i.foodId))];
   834→    const { data: foods } = await supabase.from("foods").select("id, name").in("id", foodIds);
   835→    const nameMap = new Map((foods || []).map((f: any) => [f.id, f.name]));
   836→
   837→    const items = data.items.map((i: any) => ({
   838→      saved_meal_plan_id: plan.id,
   839→      slot: (i.slot ?? "LUNCH").toUpperCase(),
   840→      food_id: i.foodId,
   841→      food_name: i.foodName ?? nameMap.get(i.foodId) ?? "Unknown",
   842→      amount: i.amount,
   843→      cal: i.cal ?? 0,
   844→      protein: i.protein ?? 0,
   845→      fat: i.fat ?? 0,
   846→      carb: i.carb ?? 0,
   847→      fiber: i.fiber ?? 0,
   848→      sodium: i.sodium ?? 0,
   849→    }));
   850→    const { error: itemsError } = await supabase.from("saved_meal_plan_items").insert(items);
   851→    if (itemsError) throw new Error(itemsError.message);
   852→  }
   853→
   854→  return plan;
   855→}
