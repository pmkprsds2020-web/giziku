   410→// ---------------------------------------------------------------------
   411→// Helper: Resolve patient ID (Prisma cuid → Supabase UUID)
   412→// Frontend stores Prisma cuid IDs in its state. Supabase has UUIDs.
   413→// This function tries: 1) Direct UUID lookup, 2) MRN lookup, 3) Prisma fallback
   414→// Returns the Supabase UUID if found, or the original ID if not.
   415→// ---------------------------------------------------------------------
   416→const patientIdCache = new Map<string, string>();
   417→
   418→export async function resolvePatientId(patientId: string): Promise<string> {
   419→  // Check if it's already a valid UUID (36 chars with dashes)
   420→  if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(patientId)) {
   421→    return patientId;
   422→  }
   423→
   424→  // Check cache
   425→  if (patientIdCache.has(patientId)) {
   426→    return patientIdCache.get(patientId)!;
   427→  }
   428→
   429→  try {
   430→    // Try Prisma first — get the MRN
   431→    const { db } = await import("@/lib/db");
   432→    const prismaPatient = await db.patient.findUnique({
   433→      where: { id: patientId },
   434→      select: { mrn: true },
   435→    });
   436→
   437→    if (!prismaPatient) {
   438→      return patientId; // Can't resolve, return original
   439→    }
   440→
   441→    // Look up in Supabase by MRN
   442→    const { client } = await getServerClient();
   443→    const { data: supaPatient } = await client
   444→      .from("patients")
   445→      .select("id")
   446→      .eq("mrn", prismaPatient.mrn)
   447→      .maybeSingle();
   448→
   449→    if (supaPatient?.id) {
   450→      patientIdCache.set(patientId, supaPatient.id);
   451→      return supaPatient.id;
   452→    }
   453→  } catch (e) {
   454→    console.warn("[resolvePatientId] Failed to resolve:", patientId, e);
   455→  }
   456→
   457→  return patientId; // Return original if resolution fails
   458→}

   461→// Helper: Resolve food ID (Prisma cuid → Supabase UUID)
   462→// ---------------------------------------------------------------------
   463→const foodIdCache = new Map<string, string>();
   464→
   465→export async function resolveFoodId(foodId: string): Promise<string> {
   466→  // Check if it's already a valid UUID
   467→  if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(foodId)) {
   468→    return foodId;
   469→  }
   470→
   471→  if (foodIdCache.has(foodId)) {
   472→    return foodIdCache.get(foodId)!;
   473→  }
   474→
   475→  try {
   476→    const { db } = await import("@/lib/db");
   477→    const prismaFood = await db.food.findUnique({
   478→      where: { id: foodId },
   479→      select: { name: true },
   480→    });
   481→
   482→    if (!prismaFood) return foodId;
   483→
   484→    const { client } = await getServerClient();
   485→    const { data: supaFood } = await client
   486→      .from("foods")
   487→      .select("id")
   488→      .eq("name", prismaFood.name)
   489→      .maybeSingle();
   490→
   491→    if (supaFood?.id) {
   492→      foodIdCache.set(foodId, supaFood.id);
   493→      return supaFood.id;
   494→    }
   495→  } catch (e) {
   496→    console.warn("[resolveFoodId] Failed to resolve:", foodId, e);
   497→  }
   498→
   499→  return foodId;
   500→}

   502→// ---------------------------------------------------------------------

   505→export async function supabaseListFoods(params?: {
   506→  search?: string;
   507→  categorySlug?: string;
   508→  limit?: number;
   509→  offset?: number;
   510→}): Promise<{ data: any[]; total: number }> {
   511→  const { client } = await getServerClient();
   512→  const limit = params?.limit ?? 100;
   513→  const offset = params?.offset ?? 0;
   514→
   515→  let query = client
   516→    .from("foods")
   517→    .select("*, food_categories!foods_category_id_fkey(*)", { count: "exact" })
   518→    .is("deleted_at", null)
   519→    .eq("approved", true)
   520→    .range(offset, offset + limit - 1)
   521→    .order("name");
   522→
   523→  if (params?.search) {
   524→    query = query.or(`name.ilike.%${params.search}%,english_name.ilike.%${params.search}%,tags.ilike.%${params.search}%`);
   525→  }
   526→
   527→  // Category filter requires subquery — skip for now, filter in memory
   528→  const { data, error, count } = await query;
   529→
   530→  if (error) {
   531→    console.error("[Supabase] listFoods error:", error);
   532→    return { data: [], total: 0 };
   533→  }
   534→
   535→  let rows = (data || []).map((row: any) =>
   536→    foodFromSupabase(row, row.food_categories),
   537→  );
   538→
   539→  // In-memory category filter
   540→  if (params?.categorySlug && params.categorySlug !== "all") {
   541→    rows = rows.filter((f: any) => f.category?.slug === params.categorySlug);
   542→  }
   543→
   544→  return { data: rows, total: count ?? rows.length };
   545→}

   547→export async function supabaseGetFood(id: string): Promise<any | null> {
   548→  const { client } = await getServerClient();
   549→  const { data, error } = await client
   550→    .from("foods")
   551→    .select("*, food_categories!foods_category_id_fkey(*)")
   552→    .eq("id", id)
   553→    .single();
   554→
   555→  if (error) {
   556→    console.error("[Supabase] getFood error:", error);
   557→    return null;
   558→  }
   559→
   560→  return foodFromSupabase(data, data.food_categories);
   561→}

   563→export async function supabaseUpsertFood(food: any): Promise<{ data: any | null; error: string | null }> {
   564→  const { client, session } = await getServerClient();
   565→  if (!session) return { data: null, error: "Authentication required. Please log in." };
   566→
   567→  const payload = foodToSupabase(food);

   776→export async function supabaseCreateMealPlan(plan: any, items: any[]): Promise<{ data: any | null; error: string | null }> {
   777→  const { client, session } = await getServerClient();
   778→  if (!session) {
   779→    console.warn("[Supabase] createMealPlan BLOCKED: No authenticated session");
   780→    return { data: null, error: "Authentication required. Please log in to save meal plans." };
   781→  }
   782→
   783→  const planPayload = mealPlanToSupabase(plan);
   784→  delete planPayload.id;
   785→
   786→  console.log("[Supabase] createMealPlan INSERT start", { table: "meal_plans", payload: planPayload, itemCount: items.length });
   787→  const { data: planData, error: planError } = await client
   788→    .from("meal_plans")
   789→    .insert(planPayload)
   790→    .select()
   791→    .single();
   792→  console.log("[Supabase] createMealPlan INSERT result", { data: planData ? { id: planData.id } : null, error: planError ? { code: planError.code, message: planError.message } : null });
   793→
   794→  if (planError) {
   795→    console.error("[Supabase] createMealPlan FAILED:", { code: planError.code, message: planError.message, details: planError.details, hint: planError.hint, payload: planPayload });
   796→    return { data: null, error: planError.message };
   797→  }
   798→
   799→  // Insert items
   800→  if (items.length > 0) {
   801→    const itemPayloads = items.map((i) => {
   802→      const payload = mealItemToSupabase(i);
   803→      payload.meal_plan_id = planData.id;
   804→      delete payload.id;
   805→      return payload;
   806→    });
   807→
   808→    const { error: itemsError } = await client
   809→      .from("meal_plan_items")
   810→      .insert(itemPayloads);
   811→
   812→    if (itemsError) {
   813→      console.error("[Supabase] createMealPlanItems error:", itemsError);
   814→      // Plan was created but items failed — return partial success
   815→      return { data: mealPlanFromSupabase(planData, []), error: `Plan created but items failed: ${itemsError.message}` };
   816→    }
   817→  }
   818→
   819→  // Fetch complete plan with items
   820→  const { data: fullPlan } = await client
   821→    .from("meal_plans")
   822→    .select("*, meal_plan_items(*, foods!meal_plan_items_food_id_fkey(*, food_categories!foods_category_id_fkey(*)))")
   823→    .eq("id", planData.id)
   824→    .single();
   825→
   826→  const planItems = (fullPlan?.meal_plan_items || []).map((item: any) =>
   827→    mealItemFromSupabase(item, item.foods ? foodFromSupabase(item.foods, item.foods.food_categories) : null),
   828→  );
   829→
   830→  return { data: mealPlanFromSupabase(fullPlan || planData, planItems), error: null };
   831→}
