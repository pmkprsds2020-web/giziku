   147→    }

   149→      await generateMut.mutateAsync({

   154→        activePresetName

   260→                  <>
   261→                    <p className="px-2 py-1 text-[10px] font-bold uppercase tracking-wider text-muted-foreground">

   264→                    {templatePresets.map((p: any) => (
   265→                      <SelectItem key={p.id} value={p.id}>

   268→                    ))}
   269→                  </>
   270→                )}
   271→              </SelectContent>
   272→            </Select>

   274→              <Badge variant="default" className="w-fit gap-1">

   277→              </Badge>
   278→            )}
   279→          </div>

   283→      {!selectedPatientId ? (

   285→          title="Pilih pasien untuk memulai"
   286→          description="Sistem akan menghitung target kalori CareLivia lalu mengoptimasi menu dari database."
   287→          icon={Utensils}
   288→        />
   289→      ) : isLoading ? (
   290→        <Skeleton className="h-96 rounded-xl" />

   301→        />

   304→      )}
   305→    </div>

   309→function MealPlanDetail({ plan }: { plan: any }) {

   311→  const [pickerOpen, setPickerOpen] = React.useState(false);
   312→  const [pickerSlot, setPickerSlot] = React.useState<MealSlot>("BREAKFAST");
   313→  const [editingItem, setEditingItem] = React.useState<any | null>(null);
   314→  const [saveMenuSlot, setSaveMenuSlot] = React.useState<MealSlot | null>(null);
   315→  const [savedMenuPickerSlot, setSavedMenuPickerSlot] = React.useState<MealSlot | null>(null);
   316→  const [saveAllOpen, setSaveAllOpen] = React.useState(false);
   317→
   318→  const addMut = useAddMealItem(plan.id);
   319→  const updateMut = useUpdateMealItem(plan.id);
   320→  const deleteMut = useDeleteMealItem(plan.id);
   321→  const createSavedMenuMut = useCreateSavedMenu();
   322→  const markSavedMenuUsedMut = useMarkSavedMenuUsed();
   323→  const createSavedMealPlanMut = useCreateSavedMealPlan();
   324→
   325→  const itemsBySlot = (plan.items || []).reduce(
   326→    (acc: Record<string, any[]>, item: any) => {
   327→      (acc[item.slot] = acc[item.slot] || []).push(item);
   328→      return acc;
   329→    },
   330→    {} as Record<string, any[]>,
   331→  );
   332→
   333→  const totalPlanCal = (plan.items || []).reduce((s: number, i: any) => s + i.cal, 0) || plan.totalCal;
   334→
   335→  const targetMet = {
   336→    cal: plan.totalCal >= plan.targetCal * 0.9 && plan.totalCal <= plan.targetCal * 1.1,
   337→    protein: plan.totalProtein >= plan.targetProtein * 0.85,
   338→    fiber: plan.totalFiber >= plan.targetFiber * 0.8,
   339→    sodium: plan.totalSodium <= plan.targetSodium,
   340→  };
   341→
   342→  const openAddPicker = (slot: MealSlot) => {
   343→    setEditingItem(null);
   344→    setPickerSlot(slot);
   345→    setPickerOpen(true);
   346→  };

   348→  const openReplacePicker = (item: any) => {
   349→    setEditingItem(item);
   350→    setPickerSlot(item.slot);
   351→    setPickerOpen(true);
   352→  };
   353→
   354→  const handleDelete = async (item: any) => {
   355→    try {
   356→      await deleteMut.mutateAsync(item.id);
   357→      toast.success(`${item.food.name} dihapus dari ${SLOT_LABELS[item.slot as MealSlot]}`);
   358→    } catch (e: any) {
   359→      toast.error(e.message || "Gagal menghapus item");
   360→    }
   361→  };
   362→
   363→  return (
   364→    <div className="space-y-4">
   365→      {/* Clinical context banner — activity/stress from assessment */}
   366→      <ClinicalContextBanner plan={plan} />
   367→
   368→      {/* Preset badge */}
   369→      {plan.preset && (
   370→        <div className="flex items-center gap-2 rounded-lg border border-border/60 bg-muted/20 p-3">
   371→          <div
   372→            className="flex h-7 w-7 items-center justify-center rounded-md text-white"
   373→            style={{ backgroundColor: plan.preset.color }}
   374→          >
   375→            <Bookmark className="h-3.5 w-3.5" />
   376→          </div>
   377→          <div className="flex-1">
   378→            <p className="text-xs font-semibold text-foreground">
   379→              Meal Plan menggunakan Preset: {plan.preset.name}
   380→            </p>
   381→            <p className="text-[11px] text-muted-foreground">
   382→              {plan.preset.description || `${Math.round(plan.preset.totalCal)} kcal · P${Math.round(plan.preset.proteinG)}g K${Math.round(plan.preset.carbG)}g L${Math.round(plan.preset.fatG)}g`}
   383→            </p>
   384→          </div>
   385→          <Badge variant="outline" className="text-[10px]">

   394→        <div className="flex-1">

   928→            Cari bahan dari database TKPI/DKBM. Nutrisi dihitung otomatis per gram.
   929→          </DialogDescription>
   930→        </DialogHeader>
   931→
   932→        <div className="space-y-3 py-1">
   933→          {/* Search */}
   934→          <div className="relative">
   935→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   936→            <Input
   937→              placeholder="Cari bahan makanan..."
   938→              value={search}
   939→              onChange={(e) => setSearch(e.target.value)}
   940→              className="pl-9"
   941→              autoFocus
   942→            />
   943→            {search && (
   944→              <button
   945→                onClick={() => setSearch("")}
   946→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"

   948→                <X className="h-4 w-4" />

   956→              <div className="flex items-start justify-between gap-2">
   957→                <div>
   958→                  <p className="text-sm font-semibold text-foreground">
