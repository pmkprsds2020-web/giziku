   429→export async function getServerClient(): Promise<{
   430→  client: SupabaseClient;
   431→  session: { user: any } | null;
   432→}> {
   433→  const client = await createClient();
   434→  const { data: { session } } = await client.auth.getSession();
   435→  return {
   436→    client,
   437→    session: session ? { user: session.user } : null,
   438→  };

   441→// ---------------------------------------------------------------------
   442→// Helper: Resolve patient ID (Prisma cuid → Supabase UUID)
   443→// Frontend stores Prisma cuid IDs in its state. Supabase has UUIDs.
   444→// This function tries: 1) Direct UUID lookup, 2) MRN lookup, 3) Prisma fallback
   445→// Returns the Supabase UUID if found, or the original ID if not.
   446→// ---------------------------------------------------------------------
   447→const patientIdCache = new Map<string, string>();
   448→
   449→export async function resolvePatientId(patientId: string): Promise<string> {
   450→  // Check if it's already a valid UUID (36 chars with dashes)
   451→  if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(patientId)) {
   452→    return patientId;
   453→  }
   454→
   455→  // Check cache
   456→  if (patientIdCache.has(patientId)) {
   457→    return patientIdCache.get(patientId)!;
   458→  }
   459→
   460→  try {
   461→    // Try Prisma first — get the MRN
   462→    const { db } = await import("@/lib/db");
   463→    const prismaPatient = await db.patient.findUnique({
   464→      where: { id: patientId },
   465→      select: { mrn: true, name: true },
   466→    });
   467→
   468→    if (!prismaPatient) {
   469→      console.warn(`[resolvePatientId] Patient not found in Prisma: ${patientId}`);
   470→      return patientId; // Can't resolve, return original
   471→    }
   472→
   473→    // Try Supabase lookup by MRN (this works even with anon role if patient exists)
   474→    const { client } = await getServerClient();
   475→    const { data: supaPatient, error } = await client
   476→      .from("patients")
   477→      .select("id")
   478→      .eq("mrn", prismaPatient.mrn)
   479→      .maybeSingle();
   480→
   481→    if (error) {
   482→      console.warn(`[resolvePatientId] Supabase lookup error for MRN ${prismaPatient.mrn}:`, error.message);
   483→    }
   484→
   485→    if (supaPatient?.id) {
   486→      console.log(`[resolvePatientId] Resolved ${patientId} → ${supaPatient.id} (via MRN ${prismaPatient.mrn})`);
   487→      patientIdCache.set(patientId, supaPatient.id);
   488→      return supaPatient.id;
   489→    }
   490→
   491→    console.warn(`[resolvePatientId] Patient not found in Supabase by MRN ${prismaPatient.mrn}, returning original ID`);
   492→  } catch (e) {
   493→    console.warn("[resolvePatientId] Failed to resolve:", patientId, e);
   494→  }
   495→
   496→  return patientId; // Return original if resolution fails
   497→}

   499→// ---------------------------------------------------------------------
   500→// Helper: Resolve food ID (Prisma cuid → Supabase UUID)
   501→// ---------------------------------------------------------------------
   502→const foodIdCache = new Map<string, string>();
   503→
   504→export async function resolveFoodId(foodId: string): Promise<string> {
   505→  // Check if it's already a valid UUID
   506→  if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(foodId)) {
   507→    return foodId;
   508→  }
   509→
   510→  if (foodIdCache.has(foodId)) {
   511→    return foodIdCache.get(foodId)!;
   512→  }
   513→
   514→  try {
   515→    const { db } = await import("@/lib/db");
   516→    const prismaFood = await db.food.findUnique({
   517→      where: { id: foodId },
   518→      select: { name: true },
   519→    });
   520→
   521→    if (!prismaFood) {
   522→      console.warn(`[resolveFoodId] Food not found in Prisma: ${foodId}`);
   523→      return foodId;
   524→    }
   525→
   526→    const { client } = await getServerClient();
   527→    const { data: supaFood, error } = await client
   528→      .from("foods")
   529→      .select("id")
   530→      .eq("name", prismaFood.name)
   531→      .maybeSingle();
   532→
   533→    if (error) {
   534→      console.warn(`[resolveFoodId] Supabase lookup error for food "${prismaFood.name}":`, error.message);
   535→    }
