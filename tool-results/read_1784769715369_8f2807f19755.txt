     2→// CareLivia — Supabase Data Layer (PRIMARY database)
     3→// ALL CRUD operations use supabase.from() — NO Prisma, NO SQLite.
     4→//
     5→// This module provides typed data access functions that map between
     6→// camelCase (frontend/Prisma convention) and snake_case (Supabase PostgreSQL).
     7→//
     8→// Authentication:
     9→//   - Server-side client reads user session from cookies
    10→//   - Public reads (foods, categories) work with anon role
    11→//   - Writes and patient data require authenticated session
    12→//   - If no session, writes return { error: "Authentication required" }
    13→// =====================================================================

   398→export async function getServerClient(): Promise<{
   399→  client: SupabaseClient;
   400→  session: { user: any } | null;
   401→}> {
   402→  const client = await createClient();
   403→  const { data: { session } } = await client.auth.getSession();
   404→  return {
   405→    client,
   406→    session: session ? { user: session.user } : null,
   407→  };
   408→}

   413→export async function supabaseListFoods(params?: {
   414→  search?: string;
   415→  categorySlug?: string;
   416→  limit?: number;
   417→  offset?: number;
   418→}): Promise<{ data: any[]; total: number }> {
   419→  const { client } = await getServerClient();
   420→  const limit = params?.limit ?? 100;
   421→  const offset = params?.offset ?? 0;
   422→
   423→  let query = client
   424→    .from("foods")
   425→    .select("*, food_categories!foods_category_id_fkey(*)", { count: "exact" })
   426→    .is("deleted_at", null)
   427→    .eq("approved", true)
   428→    .range(offset, offset + limit - 1)
   429→    .order("name");
   430→
   431→  if (params?.search) {
   432→    query = query.or(`name.ilike.%${params.search}%,english_name.ilike.%${params.search}%,tags.ilike.%${params.search}%`);
   433→  }
   434→
   435→  // Category filter requires subquery — skip for now, filter in memory
   436→  const { data, error, count } = await query;
   437→
   438→  if (error) {
   439→    console.error("[Supabase] listFoods error:", error);
   440→    return { data: [], total: 0 };
   441→  }
   442→
   443→  let rows = (data || []).map((row: any) =>
   444→    foodFromSupabase(row, row.food_categories),
   445→  );
   446→
   447→  // In-memory category filter
   448→  if (params?.categorySlug && params.categorySlug !== "all") {
   449→    rows = rows.filter((f: any) => f.category?.slug === params.categorySlug);
   450→  }
   451→
   452→  return { data: rows, total: count ?? rows.length };
   453→}

   468→  return foodFromSupabase(data, data.food_categories);

   471→export async function supabaseUpsertFood(food: any): Promise<{ data: any | null; error: string | null }> {
   472→  const { client, session } = await getServerClient();
   473→  if (!session) return { data: null, error: "Authentication required. Please log in." };
   474→
   475→  const payload = foodToSupabase(food);
   476→  const { data, error } = await client
   477→    .from("foods")
   478→    .upsert(payload)
   479→    .select("*, food_categories!foods_category_id_fkey(*)")
   480→    .single();
   481→
   482→  if (error) {
   483→    console.error("[Supabase] upsertFood error:", error);
   484→    return { data: null, error: error.message };
   485→  }
   486→
   487→  return { data: foodFromSupabase(data, data.food_categories), error: null };
   488→}
